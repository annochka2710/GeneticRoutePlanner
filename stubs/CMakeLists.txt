cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(GeneticAlgorithmStubChecks LANGUAGES CXX)
    enable_testing()
endif()
option(GA_CHECK_QT_STUBS "Also compile and check the Qt-dependent placeholders" OFF)
add_executable(geodesy_check checks/geodesy.cpp)
target_include_directories(geodesy_check PRIVATE include)
set_target_properties(geodesy_check PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
add_test(NAME geodesy_stubs COMMAND geodesy_check)
if(GA_CHECK_QT_STUBS)
    find_package(Qt5 5.12 REQUIRED COMPONENTS Widgets)
    add_executable(qt_stubs_check checks/qt_stubs.cpp)
    target_include_directories(qt_stubs_check PRIVATE include)
    target_link_libraries(qt_stubs_check PRIVATE Qt5::Widgets)
    set_target_properties(qt_stubs_check PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    add_test(NAME qt_stubs COMMAND qt_stubs_check)
    set_tests_properties(qt_stubs PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
